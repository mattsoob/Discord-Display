Reconnecting...'); setTimeout(() => location.reload(), 5000); }; function handleServerEvent(data) { switch (data.type) { case 'INITIAL_SYNC': renderSidebar(data.guilds); break; case 'SERVER_LIST': renderSidebar(data.guilds); break; case 'VOICE_CHANNELS_LIST': renderVoiceChannels(data.voiceChannels); break; case 'VOICE_STATE_CHANGE': console.log(`[HUD] Voice event: ${data.event} for ${data.username}`); // Auto-refresh the active voice channel list // Or you could build specific DOM animation logic here break; } } function renderSidebar(guilds) { const container = document.getElementById('servers-list'); container.innerHTML = guilds.map(guild => `
${guild.name}
`).join(''); } function fetchVoiceChannels(guildId) { socket.send(JSON.stringify({ type: 'FETCH_VOICE_CHANNELS', guildId })); } function renderVoiceChannels(channels) { const container = document.getElementById('voice-speakers'); container.innerHTML = channels.map(channel => `
🔊 ${channel.name} ${channel.members.map(member => `
${member.username} 🔊
`).join('')}
`).join(''); }
